# Демонстрационный стартовый скрипт эмулятора.
# Строки, начинающиеся с #, — комментарии, они не выполняются.

# Команды-заглушки
ls
ls -l /tmp
cd docs

# Переменные окружения: $HOME задана в Linux, $USERPROFILE — в Windows
ls $HOME $USERPROFILE
cd "${USERNAME}"
# Неизвестная переменная заменяется пустой строкой
ls $NO_SUCH_VARIABLE

# Комментарий в конце строки отбрасывается
ls -a  # показать скрытые файлы

exit
